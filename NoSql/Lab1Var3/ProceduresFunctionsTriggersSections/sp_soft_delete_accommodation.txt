CREATE OR REPLACE PROCEDURE sp_soft_delete_accommodation(
    p_accommodation_id bigint,
    p_user_id bigint
)
LANGUAGE plpgsql
AS $$
BEGIN
    UPDATE Accommodations
    SET
        is_deleted = TRUE,
        updated_at = CURRENT_TIMESTAMP,
        updated_by_user_id = p_user_id
    WHERE
        accommodation_id = p_accommodation_id;
END;
$$;
